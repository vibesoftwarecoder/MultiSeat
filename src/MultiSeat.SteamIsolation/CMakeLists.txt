cmake_minimum_required(VERSION 3.20)
project(MultiSeatSteamIsolation LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Steam isolation requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(third_party/minhook)
add_library(MultiSeatSteamHook SHARED Hook.cpp)
target_link_libraries(MultiSeatSteamHook PRIVATE minhook advapi32 shell32 user32)
add_executable(MultiSeatSteamLauncher Launcher.cpp)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set_target_properties(MultiSeatSteamHook PROPERTIES OUTPUT_NAME "MultiSeat.SteamHook" PREFIX "")
    set_target_properties(MultiSeatSteamLauncher PROPERTIES OUTPUT_NAME "MultiSeat.SteamLauncher")
else()
    set_target_properties(MultiSeatSteamHook PROPERTIES OUTPUT_NAME "MultiSeat.SteamHook32" PREFIX "")
    set_target_properties(MultiSeatSteamLauncher PROPERTIES OUTPUT_NAME "MultiSeat.SteamLauncher32")
endif()
foreach(target MultiSeatSteamHook MultiSeatSteamLauncher)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    endif()
endforeach()
option(MULTISEAT_BUILD_STEAM_FIXTURE "Build safe native isolation test fixture (no Steam needed)" OFF)
if(MULTISEAT_BUILD_STEAM_FIXTURE)
    enable_testing()
    add_executable(SteamIsolationFixture tests/Fixture.cpp)
    target_compile_definitions(SteamIsolationFixture PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(SteamIsolationFixture PRIVATE advapi32)
    add_executable(SteamIsolationPolicyTests tests/PolicyTests.cpp)
    target_compile_definitions(SteamIsolationPolicyTests PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(SteamIsolationPolicyTests PRIVATE shell32)
    add_test(NAME SteamIsolationPolicyTests COMMAND SteamIsolationPolicyTests)
endif()
